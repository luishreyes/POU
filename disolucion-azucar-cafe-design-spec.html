3.2 Coffee Surface Gradient

/* Dynamic radial gradient -- center is lighter (surface reflection) */ /* concentrationFactor ranges from 0.0 (pure coffee) to 1.0 (saturated) */ function drawCoffeeSurface(ctx, cx, cy, radius, concentrationFactor) { const grad = ctx.createRadialGradient( cx - radius * 0.15, cy - radius * 0.15, radius * 0.05, // inner (off-center highlight) cx, cy, radius // outer (full cup) ); // Interpolate colors based on concentration const darkEdge = lerpColor('#1a0f06', '#2a1a0c', concentrationFactor); const midTone = lerpColor('#3e2510', '#5c3a1e', concentrationFactor); const highlight = lerpColor('#5c3a1e', '#8b5e3c', concentrationFactor); grad.addColorStop(0.0, highlight); // center highlight grad.addColorStop(0.35, midTone); // mid-surface grad.addColorStop(0.7, darkEdge); // approaching rim grad.addColorStop(1.0, '#1a0f06'); // darkest at rim ctx.fillStyle = grad; ctx.beginPath(); ctx.arc(cx, cy, radius, 0, Math.PI * 2); ctx.fill(); }

3.3 Concentration Gradient Visualization

The concentration gradient uses a secondary radial gradient overlaid on the coffee surface with globalCompositeOperation: 'screen'. Sugar dissolves outward from the cube position, creating a local "sweet zone" that spreads over time. This is visualized as a warm color bloom emanating from each sugar particle cluster.
/* Concentration heatmap overlay per sugar source */ const heatGrad = ctx.createRadialGradient( sugarX, sugarY, 0, sugarX, sugarY, spreadRadius // grows over time ); heatGrad.addColorStop(0.0, 'rgba(200, 149, 108, 0.25)'); // #C8956C at 25% heatGrad.addColorStop(0.5, 'rgba(200, 149, 108, 0.08)'); // fading heatGrad.addColorStop(1.0, 'rgba(200, 149, 108, 0.0)'); // fully transparent ctx.globalCompositeOperation = 'screen'; ctx.fillStyle = heatGrad; ctx.beginPath(); ctx.arc(sugarX, sugarY, spreadRadius, 0, Math.PI * 2); ctx.fill(); ctx.globalCompositeOperation = 'source-over';

3.4 Convection / Stirring Flow Lines

/* Flow lines: curved paths that rotate around the center */ /* Visible only when stirring > 0 RPM */ function drawFlowLines(ctx, cx, cy, radius, angle, rpm) { const lineCount = 8; const opacity = Math.min(rpm / 300, 1.0) * 0.12; // max opacity 0.12 ctx.strokeStyle = `rgba(255, 220, 180, ${opacity})`; ctx.lineWidth = 1.5; ctx.lineCap = 'round'; // Draw spiral arms for (let i = 0; i < lineCount; i++) { const startAngle = angle + (i / lineCount) * Math.PI * 2; ctx.beginPath(); for (let r = radius * 0.15; r < radius * 0.85; r += 2) { const theta = startAngle + (r / radius) * Math.PI * 0.8; const x = cx + Math.cos(theta) * r; const y = cy + Math.sin(theta) * r; r === radius * 0.15 ? ctx.moveTo(x, y) : ctx.lineTo(x, y); } ctx.stroke(); } } /* Vortex center: a small spiral depression */ /* Draw a tighter spiral near the center with slightly higher opacity */ ctx.strokeStyle = 'rgba(26, 15, 6, 0.4)'; // darker depression ctx.lineWidth = 2;

3.5 Steam Effect

/* Steam particles rise from the coffee surface */ /* Quantity and size scale with temperature */ class SteamParticle { constructor(cx, cy, cupRadius) { // Random position within the top half of the cup const angle = Math.random() * Math.PI * 2; const dist = Math.random() * cupRadius * 0.6; this.x = cx + Math.cos(angle) * dist; this.y = cy + Math.sin(angle) * dist; this.size = 2 + Math.random() * 4; // 2-6px this.alpha = 0.08 + Math.random() * 0.12; // 0.08-0.20 this.vx = (Math.random() - 0.5) * 0.3; // slight drift this.vy = -0.5 - Math.random() * 1.0; // rise speed this.life = 1.0; } draw(ctx) { ctx.fillStyle = `rgba(220, 210, 200, ${this.alpha * this.life})`; ctx.beginPath(); ctx.arc(this.x, this.y, this.size * (1 + (1 - this.life) * 2), 0, Math.PI * 2); ctx.fill(); } } /* Temperature controls spawn rate: */ /* 60C -> 1 particle/frame, 95C -> 5 particles/frame */ const spawnRate = Math.floor((temp - 55) / 8);

12. Accessibility Considerations

  • Contrast ratios: All text colors on their respective backgrounds meet WCAG AA (4.5:1 for normal text, 3:1 for large text).
    • #E0E0E8 on #161622 = 11.2:1 (passes AAA)
    • #A0A0B8 on #161622 = 6.8:1 (passes AA)
    • #6A6A82 on #161622 = 3.6:1 (passes AA for large text / non-text elements)
    • #C8956C on #161622 = 5.2:1 (passes AA)
  • Focus indicators: All interactive elements (buttons, sliders, toggles) show a 2px solid #C8956C outline with 2px offset on :focus-visible.
  • Keyboard navigation: Tab order follows: sliders (top to bottom), playback buttons (left to right), speed buttons, toggles, then graph area.
  • ARIA labels: Canvas elements need role="img" with aria-label describing current state. Sliders use native <input type="range"> with aria-valuemin, aria-valuemax, aria-valuenow, and aria-label.
  • Reduced motion: Respect prefers-reduced-motion: reduce by pausing steam, disabling particle trails, and using instant transitions instead of animations.
@media (prefers-reduced-motion: reduce) { * { animation-duration: 0.01ms !important; transition-duration: 0.01ms !important; } /* In JS: check window.matchMedia('(prefers-reduced-motion: reduce)').matches */ /* If true: disable steam, reduce particle count to 1/4, skip drop animation */ } :focus-visible { outline: 2px solid #C8956C; outline-offset: 2px; }
